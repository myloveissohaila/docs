Webchat makes it easy to embed a bot on your website. If you self-host your website, just follow the instructions on this page.

<Note>

  If you use a website-building tool like [WordPress](/webchat/integrations/wordpress), [Wix](/webchat/integrations/wix) or [Webflow](/webchat/integrations/webflow), see the other pages in this section for instructions on how to add your bot.

</Note>

<Info>
  You will need:
  - A [published bot](/learn/get-started/quick-start)
  - Basic familiarity with HTML
</Info>

<Steps titleSize='h2'>
  <Step title="Get your embed code">

    <GetEmbedCode/>

  </Step>
  <Step title="Add to your website">
    Next, add the Webchat embed code to the `head` section of your website's HTML. For example:

    ```html index.html {4-5}
    <!DOCTYPE html>
    <html>
    <head>
      <script src="https://cdn.botpress.cloud/webchat/vX.X/inject.js"></script>
      <script src="https://files.bpcontent.cloud/xxxx/xx/xx/xx/xxxxxxxxxxxxx-xxxxxxxx.js" defer></script>
    </head>
    <body>
      <!-- Website content -->
    </body>
    </html>
    ```
  </Step>
  <Step>
    You're all set! Open your website and start chatting with your bot. It should look something like this:

    <Frame>
      <img
        alt="Webchat"
        className="block dark:hidden"
        src="./assets/webchat.png"
      />
      <img
        alt="Webchat"
        className="hidden dark:block"
        src="./assets/webchat-dark.png"
      />
    </Frame>
  </Step>
</Steps>

<Check>
  Done! Your bot is now live on your website.
</Check>

// ——— جستجوی خیلی ساده بر اساس امتیاز شباهت واژگانی ———
function scoreDoc(doc, query) {
  const q = query.toLowerCase();
  const t = (doc.title + " " + doc.text).toLowerCase();
  // امتیاز خام: تعداد کلمات مشترک (خیلی ساده/آموزشی)
  const qWords = [...new Set(q.split(/\s+/).filter(Boolean))];
  return qWords.reduce((acc, w) => acc + (t.includes(w) ? 1 : 0), 0);
}

function topK(docs, query, k = 2) {
  return docs
    .map(d => ({ ...d, score: scoreDoc(d, query) }))
    .sort((a, b) => b.score - a.score)
    .slice(0, k);
}

async function answerFromDocs(question) {
  const contextDocs = topK(documents, question, 2);
  const contextText = contextDocs
    .map(d => `# ${d.title}\n${d.text}`)
    .join("\n\n");

  const system = `

پاسخ کوتاه، دقیق و قابل استناد بده.`;

  const input = `
[متن زمینه]
${contextText}

[پرسش]
${question}

=> فقط بر اساس زمینه پاسخ بده و در پایان منابع را با عنوان‌ها ذکر کن.`;

  const resp = await client.responses.create({
    model: "gpt-4o-mini",
    input,
    // می‌توانید structured output یا tool calling هم اضافه کنید (رجوع به مستندات).
  });

  // خروجی متنی:
  const text = resp.output_text;
  return { text, usedDocs: contextDocs.map(d => d.title) };
}
